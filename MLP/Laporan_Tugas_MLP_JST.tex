Pada laporan tugas ini, kami menyajikan hasil penyelesaian untuk arsitektur Jaringan Saraf Tiruan (JST) dengan Multi-Layer Perceptron (1 Hidden Layer dan 1 Output Layer) menggunakan dataset riil.

\section{Deskripsi Dataset dan Arsitektur Jaringan}

\begin{itemize}
    \item \textbf{Dataset:} Menggunakan dataset \textit{Iris.csv} berbentuk numerik yang diambil dari repositori Kaggle. Sesuai instruksi, kami mengambil total 50 baris data yang terdiri dari 25 baris dari kelas \textit{Iris-versicolor} (Kelas 2) dan 25 baris dari kelas \textit{Iris-virginica} (Kelas 3). Berikut adalah contoh 10 baris cuplikan data yang mewakili kedua kelas tersebut:

    \item \textbf{Fitur (X):} Terdapat 4 fitur yaitu $X_1$ (\textit{SepalLength}), $X_2$ (\textit{SepalWidth}), $X_3$ (\textit{PetalLength}), dan $X_4$ (\textit{PetalWidth}).
    \item \textbf{Target (Y):} Kelas di-encode menggunakan \textit{one-hot encoding}. Kelas 2 merepresentasikan output Neuron 1 aktif \texttt{[1, 0]}, sedangkan Kelas 3 merepresentasikan output Neuron 2 aktif \texttt{[0, 1]}.
    \item \textbf{Arsitektur JST:}
    \begin{itemize}
        \item \textbf{Layer Input:} 4 neuron input + 1 unit bias
        \item \textbf{Hidden Layer:} 1 layer dengan 2 neuron + 1 unit bias
        \item \textbf{Layer Output:} 1 layer dengan 2 neuron (untuk kelas 2 dan kelas 3)
        \item \textbf{Fungsi Aktivasi:} \textit{Sigmoid} $f(x) = \frac{1}{1 + e^{-x}}$ diterapkan pada \textit{hidden layer} dan \textit{output layer}.
    \end{itemize}
\end{itemize}

\section{Perhitungan Manual Feedforward dan Backpropagation}

Berikut adalah rincian perhitungan matematika langkah demi langkah untuk 1 iterasi / baris data pertama (\textit{Iris-versicolor}) pada \textit{Epoch} pertama, dengan \textit{Learning Rate} ($\alpha) = 0.1$.

\subsection{Inisialisasi}
\begin{itemize}
    \item \textbf{Input ($X$):} \texttt{[7.0, 3.2, 4.7, 1.4]}
    \item \textbf{Target ($Y$):} \texttt{[1, 0]} (Kategori Iris-versicolor / Kelas 2)
    \item \textbf{Bobot Input ke Hidden ($W_1$) \& Bias ($b_1$):}
    \begin{lstlisting}
W1 = [[-0.125,  0.450],
      [ 0.231,  0.098],
      [-0.343, -0.344],
      [-0.441,  0.366]]
b1 = [[ 0.101,  0.208]]
    \end{lstlisting}
    \item \textbf{Bobot Hidden ke Output ($W_2$) \& Bias ($b_2$):}
    \begin{lstlisting}
W2 = [[-0.479,  0.469],
      [ 0.332, -0.287]]
b2 = [[-0.318, -0.316]]
    \end{lstlisting}
\end{itemize}

\subsection{Proses Feedforward}
\textbf{1. Menghitung Input pada Hidden Layer ($z_1$)} \\
Rumus: $z_1 = (X \cdot W_1) + b_1$ \\
$z_{1,1} = (7.0 \times -0.125) + (3.2 \times 0.231) + (4.7 \times -0.343) + (1.4 \times -0.441) + 0.101 = -2.270$ \\
$z_{1,2} = (7.0 \times 0.450) + (3.2 \times 0.098) + (4.7 \times -0.344) + (1.4 \times 0.366) + 0.208 = 2.574$

\vspace{0.3cm}
\textbf{2. Menghitung Output pada Hidden Layer ($a_1$) (Sigmoid)} \\
Rumus: $a_1 = \frac{1}{1 + e^{-z_1}}$ \\
$a_{1,1} = \frac{1}{1 + e^{2.270}} = 0.0936$ \\
$a_{1,2} = \frac{1}{1 + e^{-2.574}} = 0.9292$

\vspace{0.3cm}
\textbf{3. Menghitung Input pada Output Layer ($z_2$)} \\
Rumus: $z_2 = (a_1 \cdot W_2) + b_2$ \\
$z_{2,1} = (0.0936 \times -0.479) + (0.9292 \times 0.332) + (-0.318) = -0.0541$ \\
$z_{2,2} = (0.0936 \times 0.469) + (0.9292 \times -0.287) + (-0.316) = -0.5398$

\vspace{0.3cm}
\textbf{4. Menghitung Final Output ($a_2$) (Sigmoid)} \\
Rumus: $a_2 = \frac{1}{1 + e^{-z_2}}$ \\
$a_{2,1} = \frac{1}{1 + e^{0.0541}} = 0.4864$ \\
$a_{2,2} = \frac{1}{1 + e^{0.5398}} = 0.3682$ \\
\textit{Output prediksi model saat ini adalah $[0.4864, 0.3682]$, masih memiliki error terhadap target $[1, 0]$.}

\newpage
\subsection{Proses Backpropagation}

\textbf{1. Menghitung Error dan Delta pada Output Layer ($\delta_2$)} \\
$Error = Target - a_2$ \\
$Error_1 = 1 - 0.4864 = 0.5135$ \\
$Error_2 = 0 - 0.3682 = -0.3682$

\vspace{0.2cm}
$\delta_2 = Error \times a_2 \times (1 - a_2)$ \\
$\delta_{2,1} = 0.5135 \times 0.4864 \times 0.5136 = 0.1282$ \\
$\delta_{2,2} = -0.3682 \times 0.3682 \times 0.6318 = -0.0856$

\vspace{0.4cm}
\textbf{2. Menghitung Error dan Delta pada Hidden Layer ($\delta_1$)} \\
$Error_{\text{hidden}} = \delta_2 \cdot W_2^T$ \\
$Error_{\text{hidden},1} = (0.1282 \times -0.479) + (-0.0856 \times 0.469) = -0.1017$ \\
$Error_{\text{hidden},2} = (0.1282 \times 0.332) + (-0.0856 \times -0.287) = 0.0672$

\vspace{0.2cm}
$\delta_1 = Error_{\text{hidden}} \times a_1 \times (1 - a_1)$ \\
$\delta_{1,1} = -0.1017 \times 0.0936 \times 0.9064 = -0.0086$ \\
$\delta_{1,2} = 0.0672 \times 0.9292 \times 0.0708 = 0.0044$

\subsection{Pembaruan Bobot dan Bias (Update Weights)}
Rumus pembaruan: $W_{\text{new}} = W_{\text{old}} + (Input^T \cdot \delta \times \alpha)$

\vspace{0.3cm}
\textbf{1. Update Bobot Hidden ke Output ($W_2$) \& Bias ($b_2$)} \\
$W_{2_{\text{new}}} = W_2 + (a_1^T \cdot \delta_2 \times 0.1)$
\begin{lstlisting}
W2_new = [[-0.478,  0.469],
          [ 0.344, -0.295]]
b2_new = [[-0.305, -0.325]]
\end{lstlisting}

\vspace{0.3cm}
\textbf{2. Update Bobot Input ke Hidden ($W_1$) \& Bias ($b_1$)} \\
$W_{1_{\text{new}}} = W_1 + (X^T \cdot \delta_1 \times 0.1)$
\begin{lstlisting}
W1_new = [[-0.131,  0.453],
          [ 0.229,  0.100],
          [-0.348, -0.341],
          [-0.443,  0.366]]
b1_new = [[ 0.100,  0.208]]
\end{lstlisting}

\textit{Bobot dan bias baru inilah yang digunakan untuk baris data selanjutnya, hingga akhirnya proses selesai saat epochs / iterasi berakhir.}
